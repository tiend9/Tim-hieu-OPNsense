# VẬN HÀNH VÀ TROUBLESHOOTING SWIFT TRONG OPENSTACK

> **Phiên bản tham chiếu**: OpenStack **2026.1 Gazpacho** + Swift **2.34.x**

## I. THÊM STORAGE NODE / DISK MỚI

### 1. Thêm disk mới vào node hiện có

```bash
# Trên Storage Node — format disk mới:
mkfs.xfs -f -L "KOLLA_SWIFT_DATA" /dev/sdd
mkdir -p /srv/node/sdd

# Thêm vào /etc/fstab:
echo "/dev/sdd /srv/node/sdd xfs noatime,nodiratime,nobarrier,logbufs=8 0 2" >> /etc/fstab
mount -a
chown -R swift:swift /srv/node/sdd

# Kiểm tra mount:
df -h /srv/node/sdd
```

```bash
# Trên Proxy/Builder Node — thêm device mới vào Ring:

# Thêm vào Object Ring (zone 1 — storage01, disk mới sdd):
swift-ring-builder /etc/swift/object.builder add \
  --region 1 --zone 1 \
  --ip 172.16.0.21 --port 6200 \
  --replication-ip 172.16.0.21 --replication-port 6200 \
  --device sdd --weight 100

# Tương tự cho Container và Account Ring:
swift-ring-builder /etc/swift/container.builder add \
  --region 1 --zone 1 \
  --ip 172.16.0.21 --port 6201 \
  --replication-ip 172.16.0.21 --replication-port 6201 \
  --device sdd --weight 100

swift-ring-builder /etc/swift/account.builder add \
  --region 1 --zone 1 \
  --ip 172.16.0.21 --port 6202 \
  --replication-ip 172.16.0.21 --replication-port 6202 \
  --device sdd --weight 100
```

### 2. Thêm storage node hoàn toàn mới

```bash
# Trên Node mới (storage04) — chuẩn bị:
apt install -y swift swift-object swift-container swift-account xfsprogs

# Format disk:
mkfs.xfs -f /dev/sdb
mkfs.xfs -f /dev/sdc
mkdir -p /srv/node/sdb /srv/node/sdc
echo "/dev/sdb /srv/node/sdb xfs noatime,nodiratime,nobarrier,logbufs=8 0 2" >> /etc/fstab
echo "/dev/sdc /srv/node/sdc xfs noatime,nodiratime,nobarrier,logbufs=8 0 2" >> /etc/fstab
mount -a
chown -R swift:swift /srv/node/

# Copy config từ node cũ:
scp root@storage01:/etc/swift/swift.conf       /etc/swift/
scp root@storage01:/etc/swift/*.ring.gz         /etc/swift/
scp root@storage01:/etc/swift/object-server.conf   /etc/swift/
scp root@storage01:/etc/swift/container-server.conf /etc/swift/
scp root@storage01:/etc/swift/account-server.conf   /etc/swift/

# Chỉnh bind_ip trong config cho đúng IP node mới:
sed -i "s/172.16.0.21/172.16.0.24/g" /etc/swift/object-server.conf
sed -i "s/172.16.0.21/172.16.0.24/g" /etc/swift/container-server.conf
sed -i "s/172.16.0.21/172.16.0.24/g" /etc/swift/account-server.conf

# Start service:
systemctl enable --now swift-object swift-object-replicator swift-object-auditor
systemctl enable --now swift-container swift-container-replicator swift-container-auditor swift-container-updater
systemctl enable --now swift-account swift-account-replicator swift-account-auditor

# Trên Builder Node — thêm device mới vào ring (zone 4):
swift-ring-builder /etc/swift/object.builder add \
  --region 1 --zone 4 --ip 172.16.0.24 --port 6200 \
  --replication-ip 172.16.0.24 --replication-port 6200 \
  --device sdb --weight 100

swift-ring-builder /etc/swift/object.builder add \
  --region 1 --zone 4 --ip 172.16.0.24 --port 6200 \
  --replication-ip 172.16.0.24 --replication-port 6200 \
  --device sdc --weight 100
# (tương tự cho container và account ring)
```

### 3. Rebalance và phân phối Ring

```bash
# Rebalance cả 3 ring:
swift-ring-builder /etc/swift/object.builder    rebalance
swift-ring-builder /etc/swift/container.builder rebalance
swift-ring-builder /etc/swift/account.builder   rebalance

# Kiểm tra balance:
swift-ring-builder /etc/swift/object.builder
# balance: 0.00  ← lý tưởng

# Distribute ring mới sang TẤT CẢ node (cả proxy và storage):
for NODE in 10.0.0.11 172.16.0.21 172.16.0.22 172.16.0.23 172.16.0.24; do
  scp /etc/swift/*.ring.gz root@${NODE}:/etc/swift/
  echo "Distributed to $NODE"
done
```

### 4. Theo dõi replication hoàn thành

```bash
# Sau khi rebalance, Swift tự động rebalance partition qua replicator
# Theo dõi tiến trình:
swift-recon --replication --verbose

# Output mẫu (đang rebalance):
# [replication_time] low: 12.5, high: 45.3, avg: 28.1, total: 168.6
# [replication_last] low: 120.1, high: 1800.5, avg: 892.3

# Output mẫu (xong):
# [replication_last] low: 5.2, high: 30.1, avg: 15.4
# (replication_last < 600 giây là bình thường)

# Kiểm tra async_pending (object chưa được update vào container listing):
swift-recon --async
# async_pending: 0  ← cluster đã đồng bộ

# Xem trạng thái chi tiết từng node:
swift-recon -v 172.16.0.21 172.16.0.22 172.16.0.23 172.16.0.24 \
  --replication
```

## II. NODE DOWN — XỬ LÝ

### 1. Swift tự phục hồi — không cần can thiệp ngay

```text
Kịch bản: storage02 (Zone 2) bị down đột ngột

Trước khi down:
  Object "myfile" → replica 1 (zone1), replica 2 (zone2), replica 3 (zone3)

Sau khi storage02 down:
  - GET request: Proxy thử zone2 → timeout → fallback zone1 hoặc zone3 → OK
  - PUT request: Ghi zone1 và zone3 (đủ quorum 2/3) → 201 Created
  - Replica 2 bị thiếu: object-replicator trên zone1/zone3 phát hiện
    → tự copy replica sang storage02 khi storage02 up lại
    → hoặc tạo handoff replica tạm thời trên node khác
```

### 2. Kiểm tra node down và replication lag

```bash
# Kiểm tra node nào đang down:
swift-recon --all --verbose 2>&1 | grep -E "ERROR|Timeout|unreachable"

# Kiểm tra số partition không đủ replica (handoff partitions):
swift-recon --replication
# Nếu có nhiều handoff: cluster đang tự heal, cần đợi

# Xem async_pending tăng (object chưa được replicate):
swift-recon --async
# async_pending: 150  ← đang xử lý backlog

# Monitor realtime:
watch -n 10 swift-recon --replication --async

# Kiểm tra log replicator trên node còn sống:
journalctl -u swift-object-replicator -f | grep -E "ERROR|remote"
```

### 3. Node down tạm thời (restart/reboot)

```bash
# Khi node down và cần restart service:
systemctl start swift-object swift-object-replicator
systemctl start swift-container swift-container-replicator
systemctl start swift-account swift-account-replicator

# Sau khi node up lại, replicator tự detect partition thiếu và sync:
# Theo dõi:
journalctl -u swift-object-replicator -f

# Kiểm tra node đã join lại cluster:
swift-recon --all | grep -E "storage04|172.16.0.24"
```

### 4. Thay thế disk bị hỏng

```bash
# Bước 1: Gỡ device khỏi Ring (set weight = 0 để drain dần):
swift-ring-builder /etc/swift/object.builder set_weight \
  172.16.0.22 sdb 0

# Rebalance (Swift bắt đầu move partition ra khỏi disk này):
swift-ring-builder /etc/swift/object.builder rebalance

# Distribute ring:
for NODE in 10.0.0.11 172.16.0.2{1..3}; do
  scp /etc/swift/*.ring.gz root@${NODE}:/etc/swift/
done

# Đợi replication xong (async_pending = 0):
watch swift-recon --async

# Bước 2: Unmount disk hỏng:
umount /srv/node/sdb

# Bước 3: Xóa hẳn device khỏi ring:
swift-ring-builder /etc/swift/object.builder remove \
  172.16.0.22 sdb

swift-ring-builder /etc/swift/object.builder rebalance
# Distribute lại

# Bước 4: Thay disk mới, format và thêm vào ring:
mkfs.xfs -f /dev/sdb
mount /dev/sdb /srv/node/sdb
chown -R swift:swift /srv/node/sdb

swift-ring-builder /etc/swift/object.builder add \
  --region 1 --zone 2 --ip 172.16.0.22 --port 6200 \
  --replication-ip 172.16.0.22 --replication-port 6200 \
  --device sdb --weight 100

swift-ring-builder /etc/swift/object.builder rebalance
# Distribute lần cuối
```

## III. AUDITOR, REPLICATOR, UPDATER, EXPIRER

### 1. Tổng quan 4 background process

| Daemon               | Chạy trên       | Mục đích                                            | Tần suất mặc định |
|----------------------|-----------------|-----------------------------------------------------|-------------------|
| **object-auditor**   | Storage Node    | Đọc lại object, verify checksum — phát hiện bit-rot | Liên tục (30s)    |
| **object-replicator**| Storage Node    | Copy partition sang node khác đủ replica count      | Liên tục (30s)    |
| **object-updater**   | Storage Node    | Cập nhật container listing khi container server bận | 300s              |
| **object-expirer**   | 1 node duy nhất | Xóa object đã hết hạn (`X-Delete-At`)               | 300s              |

### 2. Object Auditor — phát hiện bit-rot

```bash
# Auditor chạy 2 chế độ:
# - quarantine_threshold: Kiểm tra nhanh, chỉ check size + header
# - deep audit: Đọc toàn bộ data, verify checksum (chậm hơn)

# Cấu hình /etc/swift/object-server.conf:
[object-auditor]
bytes_per_second = 10000000    # Giới hạn 10 MB/s để không ảnh hưởng I/O
files_per_second = 20          # Số file check per giây
log_time = 3600                # Log summary mỗi 1 giờ

# Xem log auditor:
journalctl -u swift-object-auditor | grep -E "ERROR|quarantine|pass"

# Object bị quarantine (checksum sai):
# Được move sang /srv/node/sdb/quarantined/objects/<hash>/
ls /srv/node/sdb/quarantined/objects/

# Xem thống kê quarantine trên cluster:
swift-recon --quarantined
# quarantined_objects: 0     ← lý tưởng
# quarantined_accounts: 0
# quarantined_containers: 0
```

### 3. Object Replicator — đảm bảo replica count

```bash
# Cấu hình /etc/swift/object-server.conf:
[object-replicator]
concurrency = 1                  # Số luồng replication đồng thời
rsync_timeout = 900              # Timeout cho 1 lần rsync (giây)
rsync_io_timeout = 30
run_pause = 30                   # Nghỉ giữa 2 lần chạy (giây)
handoffs_first = True            # Ưu tiên replicate handoff trước

# Xem log replicator:
journalctl -u swift-object-replicator -f

# Kiểm tra replication stats:
swift-recon --replication
# [replication_time] avg: 5.2     ← thời gian 1 cycle replication
# [replication_last] avg: 45.3    ← giây từ lần replication cuối

# Nếu replication_last > 3600: cluster đang lag — kiểm tra network/disk
```

### 4. Object Updater — đồng bộ container listing

```bash
# Khi nào cần updater:
# Container server tạm bận hoặc down → object server ghi thành công
# nhưng chưa update được vào container listing
# → object-updater lưu vào async_pending và retry sau

# Cấu hình /etc/swift/object-server.conf:
[object-updater]
interval = 300                   # Chạy mỗi 5 phút
concurrency = 1
slowdown = 0.01                  # Delay giữa các update (giảm load)

# Xem async_pending queue:
swift-recon --async
# async_pending: 0     ← lý tưởng

# Nếu async_pending tăng liên tục:
# → Kiểm tra container server:
systemctl status swift-container
journalctl -u swift-container -n 50
```

### 5. Object Expirer — xóa object hết hạn

```bash
# Object Expirer chỉ nên chạy trên 1 node (tránh conflict):
# /etc/swift/object-expirer.conf

[DEFAULT]
# Chỉ chạy trên node này
[object-expirer]
interval = 300
report_interval = 300
concurrency = 1

# Start service:
systemctl enable --now swift-object-expirer

# Test X-Delete-At (object tự xóa sau 60 giây):
EXPIRE_TIME=$(date -d "+60 seconds" +%s)
curl -s -X PUT \
  -H "X-Auth-Token: $TOKEN" \
  -H "X-Delete-At: $EXPIRE_TIME" \
  --data-binary "This will self-destruct" \
  "$SWIFT_URL/mycontainer/temp-object.txt"

# Sau 60 giây + 1 interval: object bị xóa
sleep 120
curl -sI -H "X-Auth-Token: $TOKEN" "$SWIFT_URL/mycontainer/temp-object.txt"
# HTTP 404 Not Found
```

### 6. Khi nào cluster "tự lành", khi nào cần can thiệp

```text
TỰ LÀNH (không cần can thiệp):
  - Node tạm down rồi up lại → replicator tự sync
  - Object chưa được replicate đủ replica → replicator tự copy
  - Container listing thiếu object → updater tự cập nhật
  - Object hết hạn chưa xóa → expirer tự dọn

CẦN CAN THIỆP:
  - Disk hỏng vật lý → thay disk, update ring
  - async_pending tăng liên tục không giảm → container server lỗi
  - quarantined_objects > 0 → bit-rot, cần kiểm tra nguồn
  - replication_last > 7200 → lag nghiêm trọng, kiểm tra network
  - Ring file không đồng bộ → redistribute ring.gz
```

## IV. TROUBLESHOOTING CINDER BACKUP LỖI

### 1. Kiểm tra trạng thái backup bị lỗi

```bash
source /root/admin-openrc

# Tìm backup bị lỗi:
openstack volume backup list | grep error

# Xem chi tiết backup lỗi:
openstack volume backup show <backup-id>
# Chú ý field "fail_reason"

# Xem log cinder-backup (systemd):
journalctl -u cinder-backup -n 100 --no-pager | grep -E "ERROR|Exception|Swift"

# Nếu dùng Kolla (container):
docker logs cinder_backup --tail 100 | grep -E "ERROR|Exception|Swift"
```

### 2. Lỗi 503 từ Swift — không reach được Swift

```bash
# Triệu chứng trong log:
# SwiftConnectionFailure: 503 Service Unavailable

# Bước 1: Kiểm tra Swift Proxy đang chạy:
systemctl status swift-proxy
curl -sI http://controller:8080/healthcheck
# Phải trả về HTTP 200

# Bước 2: Kiểm tra endpoint Keystone đúng:
openstack endpoint list --service object-store

# Bước 3: Test từ controller node:
source /root/admin-openrc
swift stat
# Nếu lỗi: vấn đề network hoặc auth

# Bước 4: Kiểm tra firewall:
iptables -L -n | grep 8080
# hoặc:
nc -zv controller 8080

# Bước 5: Nếu dùng Kolla — kiểm tra Swift container:
docker ps | grep swift
docker logs swift_proxy --tail 50
```

### 3. Lỗi Auth fail — 401/403

```bash
# Triệu chứng:
# SwiftAuthorizationFailure: 401 Unauthorized
# hoặc: 403 Forbidden

# Kiểm tra credential trong cinder.conf:
grep -E "backup_swift|backup_ceph" /etc/cinder/cinder.conf

# Test credential thủ công:
source /root/admin-openrc
openstack token issue

# Nếu dùng per_user auth — kiểm tra user có role đủ không:
openstack role assignment list --user <user> --project <project> --names

# Nếu dùng single_user auth — test credential:
swift --os-auth-url http://controller:5000/v3 \
      --os-username cinder \
      --os-password Tien123 \
      --os-project-name service \
      --os-user-domain-name Default \
      --os-project-domain-name Default \
      --auth-version 3 \
      stat

# Nếu lỗi: kiểm tra role của user cinder trong project service:
openstack role assignment list --user cinder --project service --names
# Phải có: admin hoặc swiftoperator
```

### 4. Lỗi container không tồn tại

```bash
# Triệu chứng:
# ContainerNotFound: 404 Not Found
# backup_swift_container = volumebackups không tồn tại

# Tạo container thủ công:
source /root/admin-openrc
swift post volumebackups

# Verify:
swift stat volumebackups
# X-Container-Object-Count: 0

# Hoặc đặt cinder tự tạo:
# /etc/cinder/cinder.conf:
# [DEFAULT]
# backup_swift_create_container_on_put = True
# (đây là hành vi mặc định trong Gazpacho)
```

### 5. Backup stuck — không chuyển sang available

```bash
# Triệu chứng: backup status = "creating" > 30 phút

# Bước 1: Kiểm tra cinder-backup đang chạy:
systemctl status cinder-backup
# hoặc: docker ps | grep cinder_backup

# Bước 2: Xem log realtime:
journalctl -u cinder-backup -f

# Bước 3: Kiểm tra message queue (RabbitMQ):
# Nếu cinder-backup không nhận được message từ cinder-api:
rabbitmqctl list_queues | grep cinder

# Bước 4: Reset backup bị stuck (Gazpacho hỗ trợ force reset):
openstack volume backup set --state error <backup-id>
openstack volume backup delete <backup-id>

# Hoặc thông qua DB (admin):
cinder-manage db purge 0   # dọn dẹp soft-deleted records

# Bước 5: Restart cinder-backup:
systemctl restart cinder-backup
```

### 6. Bảng tổng hợp lỗi Cinder Backup + Swift

| Lỗi                              | Log keyword                   | Nguyên nhân                    | Fix                                        |
|----------------------------------|-------------------------------|--------------------------------|--------------------------------------------|
| Backup status = error            | `SwiftConnectionFailure`      | Không reach Swift proxy        | Kiểm tra port 8080, firewall               |
| Backup status = error            | `401 Unauthorized`            | Sai credential hoặc hết token  | Kiểm tra password, role trong Keystone     |
| Backup stuck creating            | Không có log mới              | cinder-backup down hoặc MQ lỗi | Restart cinder-backup, kiểm tra RabbitMQ   |
| Restore lỗi                      | `ObjectNotFound`              | Object bị xóa khỏi Swift       | Kiểm tra `swift list volumebackups`        |
| Upload chậm                      | `timeout`                     | Network chậm hoặc Swift quá tải| Tăng `backup_swift_retry_attempts`         |
| `No space left on device`        | `ENOSPC`                      | Swift cluster full             | Thêm storage node, xóa backup cũ           |

## V. MONITORING: SWIFT-DISPERSION-REPORT

### 1. Dispersion là gì?

**swift-dispersion** là công cụ đo độ **phân tán dữ liệu** trong cluster:

- Upload test object vào **mọi partition** của ring.
- Sau đó kiểm tra xem mỗi partition có đủ replica không.
- Phát hiện sớm node mất đồng bộ **trước khi** ảnh hưởng đến user data.

### 2. Cấu hình dispersion

```bash
# Cấu hình /etc/swift/dispersion.conf:
[dispersion]
auth_url = http://controller:5000/v3
auth_user = service:swift             # project:user
auth_key = Tien123
auth_version = 3
endpoint_type = internalURL

# Container và object prefix để test:
container_populate = true
object_populate = true

# Số luồng populate:
concurrency = 25
```

### 3. Populate test data

```bash
# Lần đầu: populate test object vào tất cả partition (chạy một lần):
swift-dispersion-populate

# Output mẫu:
# Took 45.23 seconds to populate 1024 containers
# Took 120.5 seconds to populate 1024 objects

# Dispersion tạo container: .dispersion_<hash>
# Và object: .dispersion_<hash> trong mỗi partition
```

### 4. Chạy dispersion report

```bash
# Chạy report:
swift-dispersion-report

# Output mẫu (cluster tốt):
# Queried 1024 containers for dispersion reporting, 0s, 0 retries
# 100.00% of container copies found (3072 of 3072)
# Sample represents 100.00% of the container partition space
#
# Queried 1024 objects for dispersion reporting, 2s, 0 retries
# 100.00% of object copies found (3072 of 3072)
# Sample represents 100.00% of the object partition space

# Output mẫu (cluster đang mất replica):
# 98.53% of container copies found (3027 of 3072)
# ← 45 container copies thiếu → node nào đó down hoặc chưa replicate xong
```

### 5. Tự động hóa monitoring

```bash
# Cron job chạy dispersion report mỗi giờ:
cat > /etc/cron.hourly/swift-dispersion << 'EOF'
#!/bin/bash
REPORT=$(swift-dispersion-report 2>&1)
OBJECT_PCT=$(echo "$REPORT" | grep "object copies found" | grep -oP '\d+\.\d+(?=% of object)')

if (( $(echo "$OBJECT_PCT < 99.0" | bc -l) )); then
  echo "ALERT: Swift dispersion degraded: ${OBJECT_PCT}%"
  echo "$REPORT" | mail -s "[Swift Alert] Dispersion ${OBJECT_PCT}%" admin@example.com
fi

# Log kết quả:
echo "$(date): Object dispersion: ${OBJECT_PCT}%" >> /var/log/swift/dispersion.log
EOF
chmod +x /etc/cron.hourly/swift-dispersion
```

### 6. Các lệnh monitoring tổng hợp

```bash
# swift-recon — dashboard tổng quan cluster:
swift-recon --all

# Disk usage tất cả node:
swift-recon --diskusage
# Cảnh báo nếu node nào > 80%

# Kiểm tra ring consistency (tất cả node phải có ring giống nhau):
swift-recon --md5
# Nếu khác nhau → redistribute ring file ngay

# Kiểm tra load average:
swift-recon --load

# Kiểm tra quá trình unmount:
swift-recon --unmounted

# Script check nhanh sức khỏe cluster:
cat > /usr/local/bin/swift-health.sh << 'EOF'
#!/bin/bash
echo "=== Swift Cluster Health $(date) ==="
echo ""
echo "[1] Replication Status:"
swift-recon --replication 2>&1 | grep -E "avg|low|high|async"
echo ""
echo "[2] Disk Usage:"
swift-recon --diskusage 2>&1 | grep -E "avg|low|high|MAX"
echo ""
echo "[3] Quarantined Objects:"
swift-recon --quarantined 2>&1 | grep -E "quarantine"
echo ""
echo "[4] Async Pending:"
swift-recon --async 2>&1 | grep -E "async"
echo ""
echo "[5] Ring MD5 Consistency:"
swift-recon --md5 2>&1 | grep -E "All|Inconsist"
echo ""
echo "[6] Dispersion:"
swift-dispersion-report 2>&1 | grep -E "% of.*copies found"
EOF
chmod +x /usr/local/bin/swift-health.sh

# Chạy:
/usr/local/bin/swift-health.sh
```
